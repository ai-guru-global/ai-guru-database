### 5.3 技术细节

```python
# SWE-agent 核心组件

class SWEAgent:
    """SWE-bench 专用 Agent"""
    
    def __init__(self, model="claude-3-5-sonnet"):
        self.model = model
        self.codebase_index = CodebaseIndex()
        self.file_editor = FileEditor()
        self.test_runner = TestRunner()
    
    def solve(self, issue: GitHubIssue) -> Patch:
        """解决 GitHub Issue"""
        
        # 1. 理解 Issue
        repo, issue_num = parse_issue(issue)
        
        # 2. 检索相关代码
        relevant_files = self.codebase_index.retrieve(
            issue.description
        )
        
        # 3. 定位并修改
        for _ in range(self.max_attempts):
            # 读取文件
            content = read(relevant_files[0])
            
            # 分析问题
            analysis = self.analyze(content, issue)
            
            # 生成修复
            patch = self.generate_fix(content, analysis)
            
            # 测试验证
            if self.test_runner.run(patch):
                return patch
            else:
                # 分析错误，迭代修复
                error = self.test_runner.get_error()
                patch = self.iterate(patch, error)
        
        return FailedPatch()
```
