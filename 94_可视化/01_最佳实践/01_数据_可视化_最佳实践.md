# 数据可视化最佳实践 (Data Visualization Best Practices)

> 中文简称：数据可视化最佳实践

> 数据可视化最佳实践——从图表选择到配色方案、从静态出图到交互式仪表盘，让数据讲故事。

---

## 1. 图表选择指南

| 数据关系 | 推荐图表 | 适用场景 |
|---------|---------|---------|
| 分布 | 直方图、箱线图、小提琴图 | 理解数据分布特征 |
| 比较 | 柱状图、雷达图、热力图 | 多组/多指标对比 |
| 趋势 | 折线图、面积图 | 时间序列、训练曲线 |
| 组成 | 饼图、堆叠柱状图、树图 | 占比分析 |
| 关系 | 散点图、气泡图、热力图 | 相关性分析 |
| 地理 | 地图、热力地图 | 地理分布 |
| 网络 | 力导向图、桑基图 | 关系网络、流量 |

## 3. Seaborn 统计可视化

```python
import seaborn as sns

# 设置主题
sns.set_theme(style="whitegrid", palette="muted", font_scale=1.1)

# 分布图
def plot_distributions(data, column, hue=None):
    """数据分布可视化（直方图 + KDE + 箱线图）。"""
    fig, axes = plt.subplots(1, 3, figsize=(15, 4))
    
    # 直方图
    sns.histplot(data=data, x=column, hue=hue, kde=True, ax=axes[0])
    axes[0].set_title('Histogram + KDE')
    
    # 箱线图
    sns.boxplot(data=data, x=hue, y=column, ax=axes[1])
    axes[1].set_title('Box Plot')
    
    # 小提琴图
    sns.violinplot(data=data, x=hue, y=column, ax=axes[2])
    axes[2].set_title('Violin Plot')
    
    plt.tight_layout()
    plt.savefig('distributions.png', dpi=150, bbox_inches='tight')
    plt.show()

# 相关性热力图
def plot_correlation_heatmap(df, method='pearson'):
    """相关性矩阵热力图。"""
    corr = df.corr(method=method)
    mask = np.triu(np.ones_like(corr, dtype=bool))
    
    plt.figure(figsize=(10, 8))
    sns.heatmap(corr, mask=mask, annot=True, fmt='.2f', cmap='RdBu_r',
                center=0, square=True, linewidths=0.5)
    plt.title(f'{method.title()} Correlation Matrix')
    plt.tight_layout()
    plt.savefig('correlation_heatmap.png', dpi=150, bbox_inches='tight')
    plt.show()
```

## 4. Plotly 交互式可视化

```python
import plotly.express as px
import plotly.graph_objects as go
from plotly.subplots import make_subplots

# 交互式散点图
def plotly_scatter(df, x, y, color=None, size=None, hover_data=None):
    """交互式散点图。"""
    fig = px.scatter(df, x=x, y=y, color=color, size=size,
                     hover_data=hover_data,
                     title=f'{y} vs {x}',
                     template='plotly_white')
    fig.show()

# 交互式训练曲线
def plotly_training_curves(train_loss, val_loss, train_metric, val_metric):
    """交互式训练监控图。"""
    fig = make_subplots(rows=1, cols=2, subplot_titles=['Loss', 'Metric'])
    
    epochs = list(range(1, len(train_loss) + 1))
    
    fig.add_trace(go.Scatter(x=epochs, y=train_loss, name='Train Loss',
                             line=dict(color='#E74C3C')), row=1, col=1)
    fig.add_trace(go.Scatter(x=epochs, y=val_loss, name='Val Loss',
                             line=dict(color='#3498DB')), row=1, col=1)
    
    fig.add_trace(go.Scatter(x=epochs, y=train_metric, name='Train Metric',
                             line=dict(color='#E74C3C')), row=1, col=2)
    fig.add_trace(go.Scatter(x=epochs, y=val_metric, name='Val Metric',
                             line=dict(color='#3498DB')), row=1, col=2)
    
    fig.update_layout(template='plotly_white', height=400)
    fig.show()

# 3D 可散点图
def plotly_3d_scatter(df, x, y, z, color):
    """3D 散点图用于嵌入可视化。"""
    fig = px.scatter_3d(df, x=x, y=y, z=z, color=color,
                        title='3D Embedding Visualization')
    fig.show()
```

## 6. 出版级图表规范

```python
def create_publication_figure(data, filename='figure.png'):
    """创建符合论文要求的图表。"""
    # IEEE/ACM 论文标准
    fig, ax = plt.subplots(figsize=(3.5, 2.625))  # 单栏宽度
    
    # 绘图代码
    ax.plot(data['x'], data['y'], '-o', markersize=4, linewidth=1.5)
    
    # 标签
    ax.set_xlabel('X Label', fontsize=8)
    ax.set_ylabel('Y Label', fontsize=8)
    ax.set_title('Title', fontsize=9)
    
    # 刻度
    ax.tick_params(labelsize=7)
    
    # 图例
    ax.legend(fontsize=7, framealpha=0.9)
    
    # 保存（高DPI，紧凑布局）
    plt.tight_layout()
    plt.savefig(filename, dpi=300, bbox_inches='tight',
                pad_inches=0.05, format='png')
    plt.savefig(filename.replace('.png', '.pdf'), bbox_inches='tight',
                pad_inches=0.05, format='pdf')  # 矢量格式
    plt.show()
```

## 7. 常见错误与纠正

| 错误 | 问题 | 纠正 |
|------|------|------|
| 3D 饼图 | 人眼无法准确比较角度 | 用水平柱状图 |
| 双Y轴 | 误导比例关系 | 用两个子图 |
| 彩虹配色 | 色盲不友好 | 用顺序/发散配色 |
| 截断Y轴 | 夸大差异 | 从0开始或标注截断 |
| 过多颜色 | 视觉混乱 | 最多8种颜色 |
| 无标注 | 读者无法理解 | 添加标题、标签、图例 |
