## 附录：测试数据代码示例

```python
# 测试数据管理示例
from faker import Faker
from presidio_analyzer import AnalyzerEngine
import json

fake = Faker('zh_CN')

# 生成合成测试数据
def generate_test_cases(n=100):
    cases = []
    for _ in range(n):
        cases.append({
            "input": fake.text(max_nb_chars=200),
            "category": fake.random_element(["问答", "摘要", "翻译"]),
            "expected_quality": fake.random_int(60, 100),
        })
    return cases

# 隐私检测
analyzer = AnalyzerEngine()
def check_privacy(text):
    results = analyzer.analyze(text=text, language='zh')
    return [r.entity_type for r in results]
```
