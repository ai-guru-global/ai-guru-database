# Mobile AI Ops: 基于 Google Edge Gallery 的手拍即运维产品设计

> **产品定位**: 利用 Google Edge Gallery 生态系统，在移动设备上本地运行 Gemma4 等模型，实现"拍照即运维"——拍摄错误页面/日志，本地执行 Skills 完成诊断、分析、操作。

> **核心技术栈**: Google Edge Gallery (模型市场) + Gemma4 (On-Device LLM) + Edge TPU (推理加速) + 本地 Skills Engine

> **设计日期**: 2026-04-15

---

### 2.3 Gemma4 On-Device 规格

| 模型 | 参数 | 量化 | 内存占用 | 适用场景 |
|------|------|------|---------|---------|
| **Gemma4-2B** | 2B | INT4 | ~1.5GB | 快速响应、简单任务 |
| **Gemma4-7B** | 7B | INT4 | ~4.5GB | 复杂推理、诊断分析 |
| **Gemma4-7B** | 7B | FP16 | ~14GB | 最高精度 (高端设备) |

**性能基准** (Snapdragon 8 Elite @ 45 TOPS NPU):
```
Gemma4-2B INT4:  ~45 tok/s
Gemma4-7B INT4:  ~20 tok/s
响应延迟:         < 500ms (2B), < 1.5s (7B)
```

### 3.2 核心模块设计

```python
# ============================================================
# Mobile AI Ops 核心模块
# ============================================================

class MobileAIOpsApp:
    """
    Mobile AI Ops App (集成到 Google Edge Gallery)
    """

    def __init__(self):
        # Edge Gallery 集成
        self.edge_gallery = EdgeGallerySDK()
        self.model_manager = ModelManager()
        self.skill_engine = SkillEngine()

        # 核心能力
        self.ocr_processor = OCRProcessor()
        self.diagnosis_engine = DiagnosisEngine()
        self.action_executor = ActionExecutor()

        # 本地存储
        self.local_kb = LocalKnowledgeBase()
        self.audit_log = AuditLogger()

    async def diagnose_from_photo(
        self,
        image_path: str
    ) -> DiagnosisResult:
        """从照片诊断 (核心流程)"""

        # 1. OCR 识别
        ocr_result = await self.ocr_processor.process(image_path)

        # 2. 本地 Gemma4 推理
        diagnosis = await self.diagnosis_engine.analyze(
            ocr_result.text,
            context=self._build_context()
        )

        # 3. Skills 执行
        if diagnosis.requires_action:
            actions = await self.skill_engine.execute(
                diagnosis.recommended_skills
            )

        # 4. 审计日志
        await self.audit_log.log(
            event="diagnosis",
            input=image_path,
            result=diagnosis,
            actions=actions
        )

        return diagnosis


# ============================================================
# Skills 引擎
# ============================================================

class SkillEngine:
    """
    Skill 编排引擎
    """

    # 预置运维 Skills
    PRESET_SKILLS = {
        # 诊断类 (只读)
        "log_analyzer": {
            "description": "分析日志文件，提取错误模式",
            "permissions": ["read_logs"],
            "local_only": True,
            "gemma_prompt_template": """
                分析以下日志，识别错误模式:

                日志内容:
                {log_content}

                输出:
                1. 错误类型分类
                2. 关键错误信息
                3. 可能原因
                4. 严重程度 (1-5)
            """
        },

        "error_classifier": {
            "description": "分类错误类型，给出标准解决方案",
            "permissions": ["read_configs"],
            "local_only": True,
        },

        "metric_querier": {
            "description": "查询监控指标",
            "permissions": ["read_metrics"],
            "local_only": True,
        },

        # 操作类 (需确认)
        "service_restart": {
            "description": "重启服务",
            "permissions": ["write_services"],
            "require_confirm": True,
            "risk_level": "medium",
        },

        "config_backup": {
            "description": "备份配置文件",
            "permissions": ["write_configs"],
            "require_confirm": True,
            "risk_level": "low",
        },

        "log_cleanup": {
            "description": "清理日志文件",
            "permissions": ["write_files"],
            "require_confirm": True,
            "risk_level": "medium",
        },

        # 云端协同类
        "cloud_sync": {
            "description": "同步到企业知识库",
            "permissions": ["cloud_upload"],
            "require_confirm": True,
        }
    }

    async def execute(
        self,
        skill_name: str,
        params: Dict
    ) -> SkillResult:
        """执行 Skill"""

        skill = self.PRESET_SKILLS.get(skill_name)
        if not skill:
            raise SkillNotFoundError(f"Skill {skill_name} not found")

        # 1. 权限检查
        if not await self._check_permissions(skill):
            raise PermissionDeniedError()

        # 2. 风险评估
        risk_level = skill.get("risk_level", "low")

        # 3. 高风险 Skill 需确认
        if skill.get("require_confirm"):
            confirmed = await self._request_confirmation(skill, params)
            if not confirmed:
                return SkillResult(success=False, cancelled=True)

        # 4. 执行
        if skill.get("local_only"):
            return await self._execute_local_skill(skill, params)
        else:
            return await self._execute_cloud_skill(skill, params)
```

### 5.2 Edge Gallery 集成方式

```python
"""Google Edge Gallery 集成"""

class EdgeGalleryIntegration:
    """
    Google Edge Gallery 集成模块
    """

    # 模型选择
    RECOMMENDED_MODELS = {
        "fast": {
            "name": "gemma-4-2b-it-int4",
            "size_gb": 1.5,
            "use_case": "快速响应、简单任务"
        },
        "balanced": {
            "name": "gemma-4-7b-it-int4",
            "size_gb": 4.5,
            "use_case": "复杂推理、诊断分析"
        },
        "quality": {
            "name": "gemma-4-7b-it-fp16",
            "size_gb": 14,
            "use_case": "最高精度 (高端设备)"
        }
    }

    async def initialize(self):
        """初始化 Edge Gallery 环境"""

        # 1. 检查 Edge Gallery SDK
        if not self.edge_gallery.is_available():
            raise RuntimeError("Edge Gallery not available")

        # 2. 下载/加载 Gemma4 模型
        model = self.RECOMMENDED_MODELS["balanced"]
        if not self.model_manager.is_model_installed(model["name"]):
            await self.model_manager.download_model(model["name"])

        # 3. 初始化 Skills 引擎
        await self.skill_engine.load_presets()

        # 4. 初始化本地知识库
        await self.local_kb.load()

    async def run_inference(
        self,
        prompt: str,
        context: Dict = None
    ) -> InferenceResult:
        """运行本地推理"""

        # 选择推理设备 (TPU > GPU > CPU)
        device = self._select_best_device()

        # 构建推理请求
        request = InferenceRequest(
            model=self.current_model,
            prompt=prompt,
            context=context,
            device=device,
            max_tokens=512,
            temperature=0.1
        )

        # 执行推理
        result = await self.edge_gallery.run_inference(request)

        return result

    def _select_best_device(self) -> str:
        """选择最佳推理设备"""
        capabilities = self.edge_gallery.get_device_capabilities()

        if capabilities.has_tpu():
            return "tpu"
        elif capabilities.has_gpu():
            return "gpu"
        else:
            return "cpu"
```

```yaml
# Skill 定义 (符合 Edge Gallery Skills 规范)
skill_definition:
  id: "log_analyzer"
  name: "日志分析"
  version: "1.0.0"
  description: "分析日志文件，提取错误模式和建议"

  # 权限要求

    - "local:file:read"

  # 输入输出
  input:
    type: "object"
    properties:
      log_content:
        type: "string"
        description: "日志内容"
      max_errors:
        type: "integer"
        default: 10
    required: ["log_content"]

  output:
    type: "object"
    properties:
      error_count:
        type: "integer"
      errors:
        type: "array"
        items:
          type: "object"
          properties:
            type: "string"
            message: "string"
            severity: "integer"  # 1-5
            line: "integer"
      summary:
        type: "string"
      recommendations:
        type: "array"
        items:
          type: "string"

  # Gemma Prompt 模板
  prompt_template: |
    你是一个专业的运维工程师，负责分析日志文件。

    日志内容:
    {{log_content}}

    请分析并输出:
    1. 错误数量统计
    2. 每个错误的类型、消息、严重程度
    3. 一句话总结
    4. 修复建议 (如果有)

  # 执行配置
  execution:
    model: "gemma-4-7b-it-int4"
    max_tokens: 1024
    temperature: 0.1
    local_only: true

  # 安全配置
  security:
    allow_network: false
    allow_file_write: false
    data_retention: "session"  # session/forever/none
```

### 13.2 FAQ

**Q: 为什么选择 Google Edge Gallery 而不是直接开发 App?**
A: Edge Gallery 提供了成熟的模型管理、推理运行时、安全沙箱等基础设施，可以专注于上层应用开发而非底层能力。

**Q: Gemma4-2B 是否足够用于运维诊断?**
A: 对于常见错误模式识别和简单诊断，2B 模型足够。对于复杂根因分析，建议使用 7B 模型。

**Q: 如何保证诊断准确率?**
A: 通过 RAG 增强 + 本地知识库 + 持续的案例学习，持续优化模型效果。
