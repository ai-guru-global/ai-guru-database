### 2.2 核心组件实现

```python
"""AI Gateway 核心组件"""

from typing import Optional, List, Dict, Any
from dataclasses import dataclass, field
from enum import Enum
import asyncio

class ModelProvider(Enum):
    OPENAI = "openai"
    ANTHROPIC = "anthropic"
    AZURE = "azure"
    GOOGLE = "google"
    LOCAL = "local"
    CUSTOM = "custom"

@dataclass
class ModelConfig:
    """模型配置"""
    provider: ModelProvider
    model_name: str
    api_endpoint: str
    api_key_ref: str  # 密钥引用，不存储实际密钥
    capabilities: List[str] = field(default_factory=list)
    max_tokens: int = 4096
    latency_target_ms: int = 2000
    cost_per_1k_input: float = 0.0
    cost_per_1k_output: float = 0.0
    rate_limit_rpm: int = 1000
    rate_limit_tpm: int = 100000  # tokens per minute

@dataclass
class RoutingContext:
    """路由上下文"""
    user_id: str
    application_id: str
    tenant_id: str
    request_id: str
    intent: Optional[str] = None
    complexity_score: float = 0.5
    required_capabilities: List[str] = field(default_factory=list)
    budget_constraint: Optional[float] = None
    latency_sla_ms: int = 5000

class AIRequest:
    """AI 请求"""
    def __init__(self, prompt: str, routing_context: RoutingContext):
        self.prompt = prompt
        self.context = routing_context
        self.metadata: Dict[str, Any] = {}
        self.selected_model: Optional[ModelConfig] = None
        self.transformed_prompt = prompt

class AIResponse:
    """AI 响应"""
    def __init__(self, content: str, model: ModelConfig, latency_ms: float):
        self.content = content
        self.model_used = model
        self.latency_ms = latency_ms
        self.token_usage = {"input": 0, "output": 0}
        self.metadata: Dict[str, Any] = {}

class AIGateway:
    """AI Gateway 主类"""

    def __init__(self):
        self.models: Dict[str, ModelConfig] = {}
        self.router = IntelligentRouter()
        self.security = SecurityLayer()
        self.cache = ResponseCache()
        self.observability = ObservabilityManager()
        self.circuit_breaker = CircuitBreakerManager()

    def register_model(self, model_id: str, config: ModelConfig):
        """注册模型"""
        self.models[model_id] = config
        self.router.add_model(model_id, config)

    async def process_request(self, request: AIRequest) -> AIResponse:
        """处理 AI 请求的完整流程"""

        # 1. 安全检查
        await self.security.check(request)

        # 2. 意图识别和复杂度评估
        routing_info = await self.router.analyze(request)

        # 3. 模型选择
        selected_model = await self.router.select_model(
            request.context,
            routing_info
        )
        request.selected_model = selected_model

        # 4. 检查缓存
        cached = await self.cache.get(request)
        if cached:
            self.observability.record_hit(request, cached)
            return cached

        # 5. 发送请求 (带熔断保护)
        async with self.circuit_breaker.protect(selected_model):
            response = await self._call_model(selected_model, request)

        # 6. 缓存结果
        await self.cache.store(request, response)

        # 7. 记录可观测性数据
        self.observability.record_request(request, response)

        return response

    async def _call_model(
        self,
        model: ModelConfig,
        request: AIRequest
    ) -> AIResponse:
        """调用后端模型"""
        adapter = self._get_adapter(model.provider)
        start_time = asyncio.get_event_loop().time()

        response_content = await adapter.complete(
            endpoint=model.api_endpoint,
            prompt=request.transformed_prompt,
            params={"max_tokens": model.max_tokens}
        )

        latency_ms = (asyncio.get_event_loop().time() - start_time) * 1000

        return AIResponse(
            content=response_content,
            model=model,
            latency_ms=latency_ms
        )

    def _get_adapter(self, provider: ModelProvider):
        """获取对应的适配器"""
        adapters = {
            ModelProvider.OPENAI: OpenAIAdapter(),
            ModelProvider.ANTHROPIC: AnthropicAdapter(),
            ModelProvider.AZURE: AzureAdapter(),
            ModelProvider.GOOGLE: GoogleAdapter(),
        }
        return adapters.get(provider, OpenAIAdapter())
```

### 2.3 MCP 协议网关支持

```python
"""MCP (Model Context Protocol) 网关支持"""

class MCPGateway:
    """
    MCP Server Gateway - 支持 AI Agent 通过统一协议访问工具
    """

    def __init__(self):
        self.mcp_servers: Dict[str, MCPServerConfig] = {}
        self.tool_registry = ToolRegistry()
        self.protocol_handler = MCPProtocolHandler()

    async def register_server(
        self,
        server_id: str,
        endpoint: str,
        capabilities: List[str]
    ):
        """注册 MCP Server"""
        server = MCPServerConfig(
            server_id=server_id,
            endpoint=endpoint,
            capabilities=capabilities,
            status="healthy"
        )
        self.mcp_servers[server_id] = server

        # 发现可用工具
        tools = await self._discover_tools(endpoint)
        for tool in tools:
            self.tool_registry.register(
                tool_id=f"{server_id}:{tool.name}",
                tool_definition=tool,
                server_id=server_id
            )

    async def route_tool_request(
        self,
        agent_request: AgentRequest,
        tool_name: str
    ) -> ToolResponse:
        """路由工具请求到正确的 MCP Server"""

        # 1. 查找工具定义
        tool_def = self.tool_registry.get(tool_name)
        if not tool_def:
            raise ToolNotFoundError(f"Tool {tool_name} not found")

        # 2. 检查 Agent 权限
        await self._check_tool_permissions(
            agent_request.context,
            tool_def
        )

        # 3. 转发到对应 MCP Server
        server = self.mcp_servers.get(tool_def.server_id)
        response = await self.protocol_handler.forward(
            server.endpoint,
            "tools/call",
            {
                "name": tool_def.name,
                "arguments": agent_request.tool_args
            }
        )

        # 4. 审计日志
        await self._audit_tool_call(
            agent_request.context,
            tool_def,
            response
        )

        return response

    async def handle_agent_message(
        self,
        agent_id: str,
        message: AgentMessage
    ) -> AgentMessage:
        """处理 A2A Agent 间消息"""

        # 解析消息类型
        msg_type = self.protocol_handler.parse_message_type(message)

        if msg_type == "task_request":
            return await self._forward_task_to_agent(message)
        elif msg_type == "capability_query":
            return await self._return_capabilities(message)
        elif msg_type == "status_update":
            return await self._update_agent_status(message)

class ToolRegistry:
    """全局工具注册表"""

    def __init__(self):
        self.tools: Dict[str, ToolDefinition] = {}
        self.server_tools: Dict[str, List[str]] = {}  # server_id -> tool_ids

    def register(
        self,
        tool_id: str,
        tool_definition: ToolDefinition,
        server_id: str
    ):
        self.tools[tool_id] = tool_definition

        if server_id not in self.server_tools:
            self.server_tools[server_id] = []
        self.server_tools[server_id].append(tool_id)

    def get(self, tool_id: str) -> Optional[ToolDefinition]:
        return self.tools.get(tool_id)

    def list_by_server(self, server_id: str) -> List[ToolDefinition]:
        tool_ids = self.server_tools.get(server_id, [])
        return [self.tools[tid] for tid in tool_ids]
```

### 3.2 路由策略实现

```python
"""智能路由引擎"""

from dataclasses import dataclass
from typing import List, Optional, Tuple
import numpy as np

@dataclass
class ComplexityScore:
    """复杂度评分"""
    overall: float  # 0.0 - 1.0
    reasoning_depth: float
    context_length: float
    multi_step_indicator: float
    domain_specialty: float

@dataclass
class RoutingDecision:
    """路由决策"""
    model_id: str
    confidence: float
    reasoning: str
    alternatives: List[Tuple[str, float]]  # (model_id, score)

class IntelligentRouter:
    """智能路由引擎"""

    def __init__(self):
        self.models: Dict[str, ModelConfig] = {}
        self.intent_classifier = IntentClassifier()
        self.complexity_assessor = ComplexityAssessor()
        self.cost_optimizer = CostOptimizer()

    async def analyze(self, request: AIRequest) -> RoutingContext:
        """分析请求特征"""

        # 1. 意图分类
        intent = await self.intent_classifier.classify(request.prompt)

        # 2. 复杂度评估
        complexity = await self.complexity_assessor.assess(
            prompt=request.prompt,
            intent=intent
        )

        # 3. 更新请求上下文
        request.context.intent = intent
        request.context.complexity_score = complexity.overall
        request.context.required_capabilities = self._get_required_capabilities(
            intent,
            complexity
        )

        return request.context

    async def select_model(
        self,
        routing_context: RoutingContext,
        routing_info: RoutingContext
    ) -> ModelConfig:
        """选择最优模型"""

        candidates = []

        for model_id, model in self.models.items():
            # 过滤条件检查
            if not self._meets_requirements(model, routing_context):
                continue

            # 计算综合得分
            score = self._calculate_model_score(
                model,
                routing_context,
                routing_info
            )

            candidates.append((model_id, score))

        # 按得分排序
        candidates.sort(key=lambda x: x[1], reverse=True)

        if not candidates:
            raise NoAvailableModelError("No model meets requirements")

        # 返回最优模型
        best_model_id = candidates[0][0]
        return self.models[best_model_id]

    def _calculate_model_score(
        self,
        model: ModelConfig,
        context: RoutingContext,
        routing_info: RoutingContext
    ) -> float:
        """计算模型匹配得分"""

        scores = {
            # 能力匹配 (30%)
            "capability": self._score_capability_match(
                model, context.required_capabilities
            ) * 0.30,

            # 成本效率 (25%)
            "cost": self.cost_optimizer.score_model_cost(
                model, context.budget_constraint
            ) * 0.25,

            # 延迟表现 (20%)
            "latency": self._score_latency(
                model, context.latency_sla_ms
            ) * 0.20,

            # 当前负载 (15%)
            "load": self._score_current_load(model) * 0.15,

            # 历史成功率 (10%)
            "reliability": self._score_reliability(model) * 0.10,
        }

        return sum(scores.values())

class IntentClassifier:
    """意图分类器"""

    # 预定义意图类别
    INTENTS = {
        "simple_qa": ["greeting", "fact_query", "simple_definition"],
        "reasoning": ["analysis", "comparison", "evaluation", "problem_solving"],
        "code_generation": ["write_code", "debug", "explain_code", "refactor"],
        "creative": ["write_story", "generate_ideas", "brainstorm"],
        "extraction": ["extract_info", "summarize", "classify"],
        "multi_step": ["research", "complex_analysis", "multi_document"]
    }

    async def classify(self, prompt: str) -> str:
        """分类用户意图"""

        # 使用小型模型进行快速分类
        embedding = await self._embed(prompt)

        # 简单关键词匹配 + 向量相似度
        scores = {}
        for intent, keywords in self.INTENTS.items():
            keyword_score = self._keyword_match(prompt, keywords)
            scores[intent] = keyword_score

        # 返回最高分意图
        return max(scores, key=scores.get)

class ComplexityAssessor:
    """复杂度评估器"""

    async def assess(self, prompt: str, intent: str) -> ComplexityScore:
        """评估请求复杂度"""

        # 文本长度因素
        length_score = min(len(prompt) / 5000, 1.0)

        # 推理深度指标
        reasoning_indicators = [
            "分析", "比较", "评估", "为什么", "如何",
            "explain", "analyze", "compare", "evaluate"
        ]
        reasoning_score = min(
            sum(1 for w in reasoning_indicators if w in prompt.lower()) / 5,
            1.0
        )

        # 多步骤指标
        step_indicators = [
            "首先", "然后", "最后", "下一步",
            "first", "then", "next", "finally"
        ]
        multi_step_score = min(
            sum(1 for w in step_indicators if w in prompt.lower()) / 4,
            1.0
        )

        # 综合评分
        overall = (
            length_score * 0.2 +
            reasoning_score * 0.4 +
            multi_step_score * 0.2 +
            0.2  # 基础分
        )

        return ComplexityScore(
            overall=overall,
            reasoning_depth=reasoning_score,
            context_length=length_score,
            multi_step_indicator=multi_step_score,
            domain_specialty=0.0  # 需要领域知识判断
        )
```

```python
"""负载均衡器"""

from typing import List, Dict
import asyncio
import time

class LoadBalancer:
    """负载均衡器"""

    def __init__(self, strategy: str = "weighted_least_connections"):
        self.strategy = strategy
        self.model_states: Dict[str, ModelState] = {}
        self.lock = asyncio.Lock()

    async def select_instance(
        self,
        model_id: str
    ) -> str:
        """为模型选择最佳实例"""

        async with self.lock:
            instances = self.model_states.get(model_id, {})

            if self.strategy == "weighted_least_connections":
                return self._weighted_least_connections(instances)
            elif self.strategy == "round_robin":
                return self._round_robin(model_id, instances)
            elif self.strategy == "adaptive":
                return self._adaptive_selection(instances)

    def _weighted_least_connections(
        self,
        instances: Dict[str, ModelState]
    ) -> str:
        """加权最少连接"""

        weights = {}
        for instance_id, state in instances.items():
            # 权重 = 可用容量 / (当前连接数 + 1)
            weight = state.available_capacity / (state.active_connections + 1)
            weights[instance_id] = weight

        return max(weights, key=weights.get)

    def _round_robin(
        self,
        model_id: str,
        instances: Dict[str, ModelState]
    ) -> str:
        """轮询调度"""
        instance_ids = list(instances.keys())
        if not instance_ids:
            raise NoInstanceAvailableError()

        # 简单轮询
        self.round_robin_index = (self.round_robin_index + 1) % len(instance_ids)
        return instance_ids[self.round_robin_index]

    def _adaptive_selection(
        self,
        instances: Dict[str, ModelState]
    ) -> str:
        """自适应选择 - 综合考虑延迟、错误率、队列长度"""

        scores = {}
        for instance_id, state in instances.items():
            # 归一化各项指标
            latency_norm = 1 - (state.avg_latency_ms / 10000)
            error_rate_norm = 1 - state.error_rate
            queue_norm = 1 - (state.queue_length / 100)

            # 综合得分
            score = (
                latency_norm * 0.4 +
                error_rate_norm * 0.3 +
                queue_norm * 0.3
            )
            scores[instance_id] = score

        return max(scores, key=scores.get)

class CircuitBreaker:
    """熔断器 - 防止级联故障"""

    def __init__(
        self,
        failure_threshold: int = 5,
        recovery_timeout_seconds: int = 60,
        half_open_requests: int = 3
    ):
        self.failure_threshold = failure_threshold
        self.recovery_timeout = recovery_timeout_seconds
        self.half_open_requests = half_open_requests

        self.failure_count = 0
        self.last_failure_time: Optional[float] = None
        self.state = "closed"  # closed, open, half_open

    async def protect(self, operation):
        """保护操作执行"""

        if self.state == "open":
            if self._should_attempt_reset():
                self.state = "half_open"
            else:
                raise CircuitOpenError("Circuit is open")

        try:
            result = await operation()
            self._on_success()
            return result
        except Exception as e:
            self._on_failure()
            raise

    def _on_success(self):
        """成功处理"""
        self.failure_count = 0
        self.state = "closed"

    def _on_failure(self):
        """失败处理"""
        self.failure_count += 1
        self.last_failure_time = time.time()

        if self.failure_count >= self.failure_threshold:
            self.state = "open"

    def _should_attempt_reset(self) -> bool:
        """判断是否应该尝试恢复"""
        if self.last_failure_time is None:
            return True
        return time.time() - self.last_failure_time > self.recovery_timeout
```

```python
        """审核文本内容"""

        # 调用审核 API (OpenAI/Azure/自建)
        response = await self.moderation_api.check(text)

        categories = {}
        for category in self.CATEGORIES:
            categories[category] = {
                "flagged": getattr(response, category),

        # 识别违规类别
        violated = [k for k, v in categories.items() if v["flagged"]]

        return ModerationResult(
            is_safe=is_safe,
            categories=categories,
            violated_categories=violated,
            sanitized_text=response.sanitized_text or text
        )
```

```python
            top_k=1,
            filter={"type": "response_cache"}

        # 检查 TTL
        if self._is_expired(results[0].metadata["timestamp"]):
            return None

        return results[0].payload["response"]

                "response": response.content
            },

                "timestamp": time.time(),
```

```python
class MetricsCollector:
    """指标收集器 - 支持实时聚合"""

    def __init__(self):
        self.window_size_seconds = 60
        self.windows: Dict[str, List[MetricPoint]] = {}

    def record_latency(
        self,
        tenant_id: str,
        model: str,
        latency_ms: float
    ):
        """记录延迟"""

        key = f"{tenant_id}:{model}"
        if key not in self.windows:
            self.windows[key] = []

        self.windows[key].append(MetricPoint(
            timestamp=time.time(),
            value=latency_ms
        ))

        # 清理过期数据
        self._cleanup_window(key)

    def get_percentile(
        self,
        tenant_id: str,
        model: str,
        percentile: float
    ) -> float:
        """计算百分位延迟"""

        key = f"{tenant_id}:{model}"
        if key not in self.windows or not self.windows[key]:
            return 0.0

        values = [p.value for p in self.windows[key]]
        values.sort()

        index = int(len(values) * percentile)
        return values[min(index, len(values) - 1)]

    def get_error_rate(
        self,
        tenant_id: str,
        model: str
    ) -> float:
        """计算错误率"""

        # 简化实现
        return 0.0

        # Email

            await self._email_alert(alert)

        logger.warning("alert_sent", alert_type=alert_type.value)
```

### 7.2 配额管理系统

```python
"""多租户配额管理"""

from typing import Dict, Optional
import time

@dataclass
class TenantQuota:
    """租户配额"""
    tenant_id: str
    daily_token_limit: int
    monthly_token_limit: int
    daily_request_limit: int
    allowed_models: List[str]
    allowed_intents: List[str]
    max_concurrent_requests: int

class QuotaManager:
    """配额管理器"""

    def __init__(self):
        self.quotas: Dict[str, TenantQuota] = {}
        self.usage: Dict[str, Dict[str, int]] = {}  # tenant -> {metric: count}
        self.redis = RedisClient()

    async def check_quota(
        self,
        tenant_id: str,
        requested_tokens: int
    ) -> QuotaCheckResult:
        """检查配额"""

        quota = self.quotas.get(tenant_id)
        if not quota:
            return QuotaCheckResult(allowed=True)  # 无限制

        usage = await self._get_usage(tenant_id)

        # 检查各项配额
        checks = {
            "daily_tokens": usage["daily_tokens"] + requested_tokens <= quota.daily_token_limit,
            "daily_requests": usage["daily_requests"] + 1 <= quota.daily_request_limit,
            "concurrent": usage["concurrent_requests"] < quota.max_concurrent_requests,
        }

        all_passed = all(checks.values())

        return QuotaCheckResult(
            allowed=all_passed,
            reason=",".join([k for k, v in checks.items() if not v]),
            remaining={
                "daily_tokens": quota.daily_token_limit - usage["daily_tokens"],
                "daily_requests": quota.daily_request_limit - usage["daily_requests"]
            }
        )

    async def reserve_quota(
        self,
        tenant_id: str,
        tokens: int
    ):
        """预留配额"""

        key = f"quota:{tenant_id}:{self._get_date_key()}"
        await self.redis.hincrby(key, "tokens", tokens)
        await self.redis.hincrby(key, "requests", 1)

        # 设置过期时间 (48小时，覆盖时区问题)
        await self.redis.expire(key, 48 * 3600)

    async def _get_usage(self, tenant_id: str) -> Dict[str, int]:
        """获取当前使用量"""

        key = f"quota:{tenant_id}:{self._get_date_key()}"
        usage = await self.redis.hgetall(key)

        return {
            "daily_tokens": int(usage.get("tokens", 0)),
            "daily_requests": int(usage.get("requests", 0)),
            "concurrent_requests": 0
        }

    def _get_date_key(self) -> str:
        """获取日期键"""
        return time.strftime("%Y-%m-%d")
```
