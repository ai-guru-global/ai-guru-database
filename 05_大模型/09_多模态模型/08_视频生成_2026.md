## 1. 技术架构演进

### 1.1 从 2D 到时空

```
视频生成核心挑战:
- 图像生成: 2D 空间一致性
- 视频生成: 3D 时空一致性 (空间 + 时间)
- 额外维度: 运动连贯性、物理合理性、长时一致性

架构演进:
2022: 2D Diffusion + 时间层 (Make-A-Video/Imagen Video)
2023: 3D U-Net (AnimateDiff/ZeroScope)
2024: Diffusion Transformer (Sora/CogVideoX)
2025: DiT + Flow Matching (Kling 2.0/Wan 2.1)
2026: 统一多模态 DiT (视频/图像/3D 一体)
```

### 1.2 Diffusion Transformer (DiT) for Video

```python
import torch
import torch.nn as nn

class VideoDiT(nn.Module):
    """
    视频 Diffusion Transformer
    核心: 将视频视为 3D token 序列
    
    输入: 噪声视频 latent (T×H×W×C)
    输出: 去噪后的视频 latent
    """
    def __init__(self, config):
        super().__init__()
        self.config = config
        
        # 3D Patch Embedding (时空分块)
        # 将 (T, H, W) 分成 (t_p, h_p, w_p) 的 patch
        self.patch_embed = nn.Conv3d(
            in_channels=config.latent_channels,  # 通常 16
            out_channels=config.hidden_dim,
            kernel_size=(config.t_patch, config.h_patch, config.w_patch),
            stride=(config.t_patch, config.h_patch, config.w_patch)
        )
        
        # 时空位置编码
        self.pos_embed = nn.Parameter(
            torch.zeros(1, config.max_tokens, config.hidden_dim)
        )
        
        # Transformer 层 (带自适应归一化)
        self.blocks = nn.ModuleList([
            DiTBlock(config.hidden_dim, config.n_heads)
            for _ in range(config.n_layers)
        ])
        
        # 时间步 + 条件嵌入
        self.time_embed = TimestepEmbedding(config.hidden_dim)
        self.text_embed = TextConditionEmbedding(config)
        
        # 输出投影
        self.final_layer = FinalLayer(config)
    
    def forward(self, x_noisy, timestep, text_emb, fps=24):
        """
        x_noisy: (B, C, T, H, W) 噪声视频 latent
        timestep: (B,) 扩散时间步
        text_emb: (B, L, D) 文本条件
        """
        B, C, T, H, W = x_noisy.shape
        
        # 3D Patch 化
        x = self.patch_embed(x_noisy)  # (B, D, T', H', W')
        x = x.flatten(2).transpose(1, 2)  # (B, N, D)
        
        # 加位置编码
        x = x + self.pos_embed[:, :x.shape[1]]
        
        # 条件嵌入
        t_emb = self.time_embed(timestep)  # (B, D)
        
        # Transformer 去噪
        for block in self.blocks:
            x = block(x, t_emb, text_emb)
        
        # 输出
        x = self.final_layer(x, t_emb)
        
        # Reshape 回视频
        x = self.unpatchify(x, T, H, W)
        return x

class DiTBlock(nn.Module):
    """DiT 块: AdaLN + Self-Attention + Cross-Attention + FFN"""
    
    def __init__(self, dim, n_heads):
        super().__init__()
        # 自适应层归一化 (条件调制)
        self.ada_ln = AdaLayerNorm(dim)
        
        # 时空自注意力
        self.self_attn = nn.MultiheadAttention(dim, n_heads, batch_first=True)
        
        # 文本交叉注意力
        self.cross_attn = nn.MultiheadAttention(dim, n_heads, batch_first=True)
        
        # FFN
        self.ffn = nn.Sequential(
            nn.Linear(dim, dim * 4),
            nn.GELU(),
            nn.Linear(dim * 4, dim)
        )
    
    def forward(self, x, t_emb, text_emb):
        # AdaLN 调制
        scale, shift, gate = self.ada_ln(t_emb)
        
        # 自注意力 (时空)
        h = self.ada_norm(x, scale, shift)
        h = self.self_attn(h, h, h)[0]
        x = x + gate * h
        
        # 交叉注意力 (文本条件)
        h = self.cross_attn(x, text_emb, text_emb)[0]
        x = x + h
        
        # FFN
        x = x + self.ffn(x)
        return x
```

### 3.2 3D VAE

```python
class Video3DVAE(nn.Module):
    """
    3D VAE: 将视频压缩到 latent 空间
    
    压缩比:
    - 空间: 8× (H/8, W/8)
    - 时间: 4× (T/4)
    - 通道: 3 → 16
    
    总压缩: 3×T×H×W → 16×(T/4)×(H/8)×(W/8)
    约 128× 压缩
    """
    def __init__(self, latent_channels=16):
        super().__init__()
        self.encoder = Video3DEncoder(latent_channels)
        self.decoder = Video3DDecoder(latent_channels)
    
    def encode(self, video):
        """(B, 3, T, H, W) → (B, 16, T/4, H/8, W/8)"""
        return self.encoder(video)
    
    def decode(self, latent):
        """(B, 16, T/4, H/8, W/8) → (B, 3, T, H, W)"""
        return self.decoder(latent)
```

### 4.1 条件控制

```python
# 2026 视频生成控制方式:

控制方式 = {
    "文本控制": "自然语言描述场景/动作/风格",
    "图像控制": "首帧/尾帧/关键帧 → 视频",
    "运动控制": "光流/轨迹/骨骼 → 运动引导",
    "相机控制": "推拉摇移/环绕/跟随",
    "参考视频": "风格迁移/动作迁移",
    "3D 控制": "深度图/法线/点云 → 3D 一致",
    "音频控制": "音乐节奏 → 视频节拍 (Veo 3)",
}

# ControlNet for Video:
class VideoControlNet(nn.Module):
    """视频 ControlNet: 额外条件注入"""
    
    def __init__(self, base_model, condition_type="depth"):
        super().__init__()
        self.base = base_model
        # 条件编码器
        self.condition_encoder = ConditionEncoder(condition_type)
        # 零卷积 (初始化为0，渐进学习)
        self.zero_convs = nn.ModuleList([
            zero_init_conv(dim) for dim in base_model.channel_dims
        ])
    
    def forward(self, x_noisy, timestep, text, condition):
        # 编码条件
        cond_features = self.condition_encoder(condition)
        
        # 注入到 DiT 各层
        for i, (block, zero_conv) in enumerate(
            zip(self.base.blocks, self.zero_convs)
        ):
            x_noisy = block(x_noisy, timestep, text)
            x_noisy = x_noisy + zero_conv(cond_features[i])
        
        return x_noisy
```

```python
# 2026: 实时交互式视频生成
# 用户可以在生成过程中实时引导

class InteractiveVideoGen:
    """
    交互式视频生成:
    - 实时调整方向/风格/内容
    - 游戏化: 用户选择 → 视频分支
    - 直播: 观众投票影响视频走向
    """
    def generate_interactive(self, initial_prompt, user_inputs):
        video_frames = []
        current_state = self.encode_prompt(initial_prompt)
        
        for user_input in user_inputs:
            # 用户实时输入 (文本/手势/语音)
            current_state = self.update_state(current_state, user_input)
            # 生成下一段
            next_frames = self.generate_segment(current_state, n_frames=24)
            video_frames.extend(next_frames)
        
        return video_frames
```
