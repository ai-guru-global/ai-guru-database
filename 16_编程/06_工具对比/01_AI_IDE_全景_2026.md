### 2.2 Agent 模式工作原理

```python
class AI_IDE_Agent:
    """
    AI IDE Agent 模式核心循环:
    观察 → 思考 → 行动 → 验证 → 重复
    """
    def __init__(self, model, tools):
        self.model = model  # LLM
        self.tools = {
            "read_file": "读取文件内容",
            "write_file": "创建/修改文件",
            "search_code": "搜索代码库",
            "run_terminal": "执行终端命令",
            "list_files": "列出目录",
            "grep": "正则搜索",
            "browser": "浏览器操作",
        }
    
    async def solve_task(self, task_description):
        """Agent 主循环"""
        # 1. 规划
        plan = await self.model.generate(
            f"任务: {task_description}\n请制定实施计划"
        )
        
        # 2. 逐步执行
        for step in plan.steps:
            # 观察: 收集上下文
            context = await self.gather_context(step)
            
            # 思考: 决定行动
            action = await self.model.generate(
                f"当前步骤: {step}\n上下文: {context}\n选择工具和参数"
            )
            
            # 行动: 执行工具
            result = await self.execute_tool(action)
            
            # 验证: 检查结果
            if not self.verify(result, step):
                # 失败则重试/调整
                await self.handle_failure(step, result)
        
        # 3. 最终验证
        await self.run_tests()
        return "任务完成"
```

### 3.4 选型决策树

```python
def select_ai_ide(profile: dict) -> str:
    """2026 AI IDE 选型"""
    
    if profile["budget"] == 0:
        return "Trae (免费) 或 Windsurf (免费额度)"
    
    if profile["primary_language"] == "中文" and profile["in_china"]:
        return "Trae (中文最优+免费)"
    
    if profile["need_full_autonomy"]:
        return "Qoder 或 Devin (全自主 Agent)"
    
    if profile["prefer_terminal"]:
        return "Claude Code (CLI Agent)"
    
    if profile["team_size"] > 50:
        return "GitHub Copilot Enterprise 或 Augment"
    
    if profile["coding_intensity"] == "high":
        return "Cursor (最佳补全+Agent)"
    
    return "Cursor (综合最优)"
```
